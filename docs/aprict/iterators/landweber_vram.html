 54    @pydantic.dataclasses.dataclass(frozen=True, kw_only=True)
 55    class Config:
 56        """Configuration dataclass for the Landweber iterator."""
 57
 58        rayleigh_quotient_num_iterations: int = 10
 59        acceleration_factor: float = 1
 60        fast_backproject: bool = True
 61        soft_from_cylinder_relrad: float | None = (
 62            0.95  # percentage of the cylinder radius from which softening will occur
 63        )
 64        bp_soft_from_detedge_relrad: float | None = (
 65            0.95  # percentage of the detector half-width from which to soften
 66        )
 67        # These not implemented yet:
 68        # enforce_positivity_constraint: bool     = False
 69        # soften_to_cylinder_relhei: float | None = 0.05  # the height of the object to reserve for softening to zero. This is expressed as a proportion of the radius of the inscribed x/y cylinder.
 70
 71        # denoising: float | None = None  # Applies an approximate TV gradient descent between iterations. This is NOT accounted for in the calculation of the next step, nor the calculated and printed error metric. This denoising should be considered an approximate ad-hoc method. Suggested value range: 0 to 0.05.

 73    def __init__(
 74        self,
 75        config: Config,
 76        volume: Volume,
 77        projs: cbfp.ProjectionSet | list[cbfp.ProjectionSet],
 78        *,
 79        proj_zero_masks: cp.ndarray | list[cp.ndarray] | None = None,
 80        logger: logging.Logger | None,
 81    ) -> None:
 82        self._logger = logger if logger is not None else logging.getLogger()
 83        self._config = config
 84        self._vol = volume
 85        self._meas = projs if type(projs) is list else [projs]  # ty:ignore[invalid-assignment]  # ty is bugged
 86        if proj_zero_masks is None:
 87            self._proj_zero_masks = [None for p in self._meas]
 88        elif type(projs) is list:
 89            if type(proj_zero_masks) is not list:
 90                msg = "The input measurements were a list, but only a single mask was provided."
 91                raise ValueError(msg)
 92            self._proj_zero_masks = proj_zero_masks
 93        else:
 94            if type(proj_zero_masks) is list:
 95                msg = "The input measurements were not a list, but a list of masks was provided."
 96                raise ValueError(msg)
 97            self._proj_zero_masks = [proj_zero_masks]
 98        self._config_fp = cbfp.ConfigProject()
 99        soften_det_edges_proportion = (
100            0
101            if self._config.bp_soft_from_detedge_relrad is None
102            else (1 - self._config.bp_soft_from_detedge_relrad)
103        )
104        self._config_bp = cbfp.ConfigBackproject(
105            fast_approx=self._config.fast_backproject,
106            cyl_mask_rad=min(*volume.shape[:2]) / 2 * volume.voxwid,
107            apply_summand_multiplier=False,
108            soften_det_edges_proportion=soften_det_edges_proportion,
109        )
110        for projs in self._meas:
111            if projs.data is None:
112                msg = "A provided ProjectionSet object has no data."
113                raise ValueError(msg)
114        # Make the projection scratch space. Equal in size to one ordered subset.
115        self._proj_scratch_space = [
116            projs.zeroed_copy() for projs in self._meas
117        ]
118        self._rayleigh_quotients = self.calculate_rayleigh_quotients(
119            num_iterations=self._config.rayleigh_quotient_num_iterations
120        )
121
122        # Apply zero masks to projection data
123        for proj, zm in zip(self._meas, self._proj_zero_masks):
124            if zm is not None:
125                cp.multiply(proj.data, zm, out=proj.data)

127    def calculate_rayleigh_quotients(self, num_iterations: int) -> list[float]:
128        """Compute the list Rayleigh quotients of A^T A (one for each ordered subset) using the iterated power method."""
129        fp_config: cbfp.ConfigProject = self._config_fp
130        bp_config: cbfp.ConfigBackproject = self._config_bp
131        # Check if current volume is 0s. In that case, don't allocate new memory. Just use that one.
132        flag_using_existing_volume = self._vol.norm == 0
133        scratch_volume: Volume = (
134            self._vol
135            if flag_using_existing_volume
136            else self._vol.zeroed_copy()
137        )
138        scratch_projs_list: list[cbfp.ProjectionSet] = self._proj_scratch_space
139        for scratch_projs in scratch_projs_list:
140            if scratch_projs.data is None:
141                msg = "ProjectionSet must have its data buffer initialised."
142                raise ValueError(msg)
143
144        rayleigh_quotient_list: list[float] = []
145        scale_factor: float = 1 / float(
146            np.mean(scratch_volume.shape)
147        )  # a factor used to scale things to prevent overflow etc
148        self._logger.debug(
149            "Calculating Rayleigh quotient(s) for the ordered subset(s)..."
150        )
151        with logtools.ProgressBar(
152            desc="Calc. Rayleigh quot.:",
153            maxval=num_iterations * len(scratch_projs_list),
154            valunit="iteration",
155            logger=self._logger,
156        ) as pbar:
157            for scratch_projs_index, scratch_projs in enumerate(
158                scratch_projs_list
159            ):
160                assert (
161                    scratch_projs.data is not None
162                )  # useful assertion for type-checking
163                # Generate a random positive volume
164                v = np.abs(
165                    np.random.randn(*scratch_volume.shape) * scale_factor  # ruff: ignore[numpy-legacy-random]
166                ).astype(np.float32)
167                scratch_volume.copy_from(v)
168                # self._hardmask_vol(
169                #     scratch_volume
170                # )  # Note: not exactly the same as the cyl_mask_rad offered by the backprojection. This may matter.
171                svn = scratch_volume.norm
172                for numit in range(
173                    1, num_iterations + 1
174                ):  # Use N iterations of power method to find the rayleigh quotient
175                    # Normalize the volume
176                    scratch_volume.multiply(1 / svn)
177                    # Apply A^T A to the volume
178                    scratch_projs.data.fill(0)
179                    cbfp.project(
180                        config=fp_config,
181                        volume=scratch_volume,
182                        projs=scratch_projs,
183                        block=True,
184                    )
185                    rayleigh_quotient = cp.square(
186                        cp.linalg.norm(scratch_projs.data)
187                    )
188                    scratch_volume.fill(0)
189                    cbfp.backproject(
190                        config=bp_config,
191                        volume=scratch_volume,
192                        projs=scratch_projs,
193                        block=True,
194                        logger=self._logger,
195                    )
196                    svn = scratch_volume.norm
197                    pbar.update_to(
198                        numit + num_iterations * scratch_projs_index
199                    )
200                rayleigh_quotient_list.append(
201                    rayleigh_quotient
202                )  # Could also use svn instead of rayleigh_quotient
203        self._logger.debug("...done calculating Rayleigh quotient(s).")
204        if flag_using_existing_volume:
205            self._vol.fill(0)  # reset the volume to 0
206        del scratch_volume
207        del scratch_projs_list
208        gpu_tools.clear_cache()  # If new memory was allocated for the scratch space, then it can be released now
209        return rayleigh_quotient_list

211    def _softmask_vol(self, v: Volume) -> None:
212        rr = self._config.soft_from_cylinder_relrad
213        if rr is not None:
214            v.soften_to_xy_cylinder(rr)
215
216    def _hardmask_vol(self, v: Volume) -> None:
217        rr = 1
218        if rr is not None:
219            v.soften_to_xy_cylinder(rr)

221    def iterate(self) -> None:
222        """Perform an iteration of the algorithm."""
223        # TODO: reduce memory usage by reusing the same projection data buffer as scratch space.  # ruff: ignore[line-contains-todo]
224
225        # Loop over the ordered subsets
226        for meas, proj_scratch_space, rq, zm in zip(
227            self._meas,
228            self._proj_scratch_space,
229            self._rayleigh_quotients,
230            self._proj_zero_masks,
231        ):
232            # Put the projection residual into the projection scratch space
233            cp.copyto(src=meas.data, dst=proj_scratch_space.data)
234            config_fp = cbfp.ConfigProject(multiply_before_adding=-1)
235            cbfp.project(
236                config=config_fp,
237                volume=self._vol,
238                projs=proj_scratch_space,
239                block=True,
240            )
241            if zm is not None:
242                cp.multiply(
243                    proj_scratch_space.data, zm, out=proj_scratch_space.data
244                )
245            # Add the backprojection of the residual into the volume
246            # Divide by relaxation factor first
247            relaxation_factor = self._config.acceleration_factor / rq
248            cp.multiply(
249                proj_scratch_space.data,
250                relaxation_factor,
251                out=proj_scratch_space.data,
252            )
253            cbfp.backproject(
254                config=self._config_bp,
255                volume=self._vol,
256                projs=proj_scratch_space,
257                summand_multiplier=None,
258                block=True,
259                logger=self._logger,
260            )
261
262    @property
263    def volume(self) -> Volume:
264        """The volume/tomogram."""
265        return self._vol
