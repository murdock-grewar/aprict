418    def prune_to_bbox(
419        self,
420        xyz_min: tuple[float, float, float]
421        | np.ndarray[tuple[typing.Literal[3]], np.dtype[np.floating]],
422        xyz_max: tuple[float, float, float]
423        | np.ndarray[tuple[typing.Literal[3]], np.dtype[np.floating]],
424    ) -> FBVolume:
425        """Return a new FBVolume containing only the shapes relevant to the given bounding box."""
426        # Handle float inputs
427        xyz_min = np.array(xyz_min, dtype=np.float32)
428        xyz_max = np.array(xyz_max, dtype=np.float32)
429
430        accept_mask = np.ones(len(self.shapes), dtype=np.bool)
431        bboxes: np.ndarray[tuple[int], np.dtype[ks_bbox]] = (  # ty:ignore[invalid-type-arguments]
432            bboxes_FORBILD.bboxes_FORBILD_shapes(self.shapes, block=True).get()
433        )
434        # Prune away the pieces
435        a = np.array(
436            [b["xyz_max"]["x"] >= xyz_min[0] for b in bboxes], dtype=np.bool
437        )
438        accept_mask = np.logical_and(accept_mask, a)
439        a = np.array(
440            [b["xyz_max"]["y"] >= xyz_min[1] for b in bboxes], dtype=np.bool
441        )
442        accept_mask = np.logical_and(accept_mask, a)
443        a = np.array(
444            [b["xyz_max"]["z"] >= xyz_min[2] for b in bboxes], dtype=np.bool
445        )
446        accept_mask = np.logical_and(accept_mask, a)
447        a = np.array(
448            [b["xyz_min"]["x"] <= xyz_max[0] for b in bboxes], dtype=np.bool
449        )
450        accept_mask = np.logical_and(accept_mask, a)
451        a = np.array(
452            [b["xyz_min"]["y"] <= xyz_max[1] for b in bboxes], dtype=np.bool
453        )
454        accept_mask = np.logical_and(accept_mask, a)
455        a = np.array(
456            [b["xyz_min"]["z"] <= xyz_max[2] for b in bboxes], dtype=np.bool
457        )
458        accept_mask = np.logical_and(accept_mask, a)
459        return FBVolume(self.shapes[accept_mask])
