  45    def CustomComposite(cls) -> FBVolume:
  46        """A custom phantom stitching together and slightly modifying the Hip, Abdomen, Thorax, Head and Jaw phantom."""
  47        hi, ab, th, he, ja = [
  48            FBVolume(
  49                ppf.convert_ph_file_to_shapes(
  50                    _path_to_FORBILD_src.joinpath("custom_composite/" + fn)
  51                )
  52            )
  53            for fn in (
  54                "part_01_hip.pha",
  55                "part_02_abdomen.pha",
  56                "part_03_thorax.pha",
  57                "part_04_head.pha",
  58                "part_05_jaw.pha",
  59            )
  60        ]
  61
  62        # Create the lungs phantom which will sit on top of everything
  63        lu = PhantomContainer.LungPhantom(
  64            fix=True, generation=10, taper=True, scale_factor=0.60
  65        )
  66        lu.translate(axis="z", amount=150)
  67        lu.scale_len(
  68            1.8
  69        )  # initially too small to fit in the thorax lung pockets
  70
  71        # Convert cm to mm.
  72        ja.scale_len(10)
  73        he.scale_len(10)
  74        th.scale_len(10)
  75        ab.scale_len(10)
  76        hi.scale_len(10)
  77
  78        # The jaw is not located correctly with respect to the head. Correct it here.
  79        ja.translate(axis="y", amount=-20)
  80        ja.translate(axis="z", amount=-120)
  81        # Translate the head up and slightly forwards
  82        amount = 360
  83        he.translate(axis="z", amount=amount)
  84        ja.translate(axis="z", amount=amount)
  85        amount = 0  # Never mind. Don't translate it forwards.
  86        he.translate(axis="y", amount=amount)
  87        ja.translate(axis="y", amount=amount)
  88        # Translate abdomen and hip down
  89        amount = -200
  90        ab.translate(axis="z", amount=amount)
  91        hi.translate(axis="z", amount=amount)
  92        # Translate the hips down further
  93        amount = -300
  94        hi.translate(axis="z", amount=amount)
  95
  96        # Remove the background shapes making up the thoracical/abdominal cavity. We'll place these objects first in the list after composing the full object, so that they remoain in the background.
  97        # cavity = FBVolume(th.shapes[[2]]).superimpose_onto(FBVolume(ab.shapes[[0]]))
  98        cavity = FBVolume(th.shapes[[2]])
  99        ab = FBVolume(ab.shapes[1:])
 100        th = FBVolume(np.concatenate((th.shapes[0:2], th.shapes[3:]), axis=0))
 101        # Extend the cavity to meet the hips
 102        cavity.shapes[0]["z"] -= 50
 103        cavity.shapes[0]["l"] += 100
 104        # Superimpose in order
 105        v = ja.superimpose_onto(lu)  # jaws on top of lungs is better
 106        v = v.superimpose_onto(he)
 107        v = v.superimpose_onto(th)
 108        v = v.superimpose_onto(ab)
 109        v = v.superimpose_onto(hi)
 110        v = v.superimpose_onto(cavity)
 111        return v  # ruff: ignore[unnecessary-assign]

 262    @classmethod
 263    def _LungPhantom_phastring(  # ruff: ignore[complex-structure]
 264        cls,
 265        generation: int,
 266        scale_factor: float
 267        | None = None,  # Leave at None for default configuration
 268        taper: bool | None = None,  # Leave at None for default configuration
 269        fix: bool
 270        | None = None,  # Leave at None for default configuration. Turning this True will fix what appear to be errors in the original phantom. These errors are particularly noticeable with e.g. taper=True.
 271    ) -> str:
 272        """Generate the [Lung phantom](https://www.dkfz.de/en/x-ray-imaging-and-computed-tomography/lung-phantom) (by Markus Blank and Marc Kachelrieß) up to generation n."""
 273        if type(generation) is not int or generation < 2:
 274            msg = "'generation' must be an integer >=2"
 275            raise ValueError(msg)
 276        if scale_factor is None:
 277            scale_factor = 0.86  # default value is 0.86
 278        if taper is None:
 279            taper = False  # default phantom does not taper. It uses cylinders, not truncated cones.
 280        if fix is None:
 281            fix = False
 282        # Lindenmayer System
 283        s = "T[L[s][b]][R[s][b]]"
 284        for _ in range(generation - 2):
 285            s = s.replace("b", "B[X][Y]").replace("s", "S[X][Y]")
 286            s = s.replace("X", "b").replace("Y", "s")
 287        s = s.replace("b", "B").replace("s", "S")
 288        # Construct a dictionary tree of the terminal symbols T, L, R, B, S
 289        patt = regex.compile(r"(?V1)\[((?:[^\]\[]+|(?R))*+)\]")
 290
 291        @dataclasses.dataclass(frozen=True)
 292        class Branch:
 293            generation: int
 294            scale: float
 295            length: float
 296            taper_factor: float  # how much it shrinks by the end of the tube
 297            radius_outer: float
 298            radius_inner: float
 299            u: tuple[float, float, float]  # start point
 300            v: tuple[float, float, float]  # end point
 301            x: tuple[
 302                float, float, float
 303            ]  # defines the branch plane together with z
 304            y: tuple[
 305                float, float, float
 306            ]  # new branches will be rotated about y
 307            z: tuple[float, float, float]  # direction of branch
 308
 309        def generate_branch(symbol: str, parent: Branch | None) -> Branch:
 310            if parent is None:
 311                if symbol not in ("T", "L", "R"):
 312                    raise ValueError()
 313            match symbol:
 314                case "T":
 315                    return Branch(
 316                        0,
 317                        1.0,
 318                        100.0,
 319                        1,  # do not taper the trachea
 320                        9.0,
 321                        7.9,
 322                        (0.0, 0.0, 0.0),
 323                        (0.0, 0.0, 100.0),
 324                        (1.0, 0.0, 0.0),
 325                        (0.0, 1.0, 0.0),
 326                        (0.0, 0.0, 1.0),
 327                    )
 328                case "L":
 329                    return Branch(
 330                        1,
 331                        0.86,
 332                        34.4,
 333                        0.86 if taper else 1,
 334                        5.59,
 335                        4.644,
 336                        (0.0, 0.0, 100.0),
 337                        (28.1788, 0.0, 119.731),
 338                        (0.573576, 0.0, -0.819152),
 339                        (0.0, 1.0, 0.0),
 340                        (0.819152, 0.0, 0.573576),
 341                    )
 342                case "R":
 343                    return Branch(
 344                        1,
 345                        0.86,
 346                        43.0,
 347                        0.86 if taper else 1,
 348                        6.45,
 349                        5.504,
 350                        (0.0, 0.0, 100.0),
 351                        (-35.2235, 0.0, 124.664),
 352                        (-0.573576, 1.22461e-16, -0.819152),
 353                        (-7.02405e-17, -1.0, -1.00314e-16),
 354                        (-0.819152, 0.0, 0.573576),
 355                    )
 356                case "B":
 357                    assert parent is not None
 358                    r = scipy.spatial.transform.Rotation.from_rotvec(
 359                        np.array(parent.y) * -25, degrees=True
 360                    )
 361                    #
 362                    generation = parent.generation + 1
 363                    scale = parent.scale * scale_factor
 364                    length = scale * 50
 365                    # The amount to taper increase the radius toward the root
 366                    # depends on the radius of the parent.
 367                    # The parent can be the smaller of the two parents,
 368                    # in which case it needs to taper by less than the full
 369                    # inverse scale_factor
 370                    radius_outer = scale * 8.5 * (7.5 / 8.5 if fix else 1)
 371                    radius_inner = radius_outer * (7.4 / 8.5)
 372                    taper_factor = (
 373                        radius_outer / parent.radius_outer if taper else 1
 374                    )
 375                    u = parent.v
 376                    z = tuple(r.apply(parent.z))
 377                    x = tuple(
 378                        r.apply(parent.x)
 379                    )  # the web page isn't clear about how the new x should be determined
 380                    r2 = scipy.spatial.transform.Rotation.from_rotvec(
 381                        np.array(z) * 90,
 382                        degrees=True,  # I thought the angle here should have a minus sign, but no, it matches the 3rd generation example without it.
 383                    )
 384                    x = tuple(r2.apply(x))
 385                    v = tuple(np.array(u) + length * np.array(z))
 386                    y = tuple(np.cross(z, x))
 387                    return Branch(
 388                        generation,
 389                        scale,
 390                        length,
 391                        taper_factor,
 392                        radius_outer,
 393                        radius_inner,
 394                        u,
 395                        v,
 396                        x,
 397                        y,
 398                        z,
 399                    )
 400                case "S":
 401                    assert parent is not None
 402                    r = scipy.spatial.transform.Rotation.from_rotvec(
 403                        np.array(parent.y) * 45, degrees=True
 404                    )
 405                    #
 406                    generation = parent.generation + 1
 407                    scale = parent.scale * scale_factor
 408                    length = scale * 40
 409                    radius_outer = scale * 7.0 * (7.5 / 8.5 if fix else 1)
 410                    radius_inner = radius_outer * (5.9 / 7.0)
 411                    taper_factor = (
 412                        radius_outer / parent.radius_outer if taper else 1
 413                    )
 414                    u = parent.v
 415                    z = tuple(r.apply(parent.z))
 416                    x = tuple(
 417                        r.apply(parent.x)
 418                    )  # the web page isn't clear about how the new x should be determined
 419                    r2 = scipy.spatial.transform.Rotation.from_rotvec(
 420                        np.array(z) * 10,
 421                        degrees=True,  # I thought the angle here should have a minus sign, but no, it matches the 3rd generation example without it.
 422                    )
 423                    x = tuple(r2.apply(x))
 424                    v = tuple(np.array(u) + length * np.array(z))
 425                    y = tuple(np.cross(z, x))
 426                    return Branch(
 427                        generation,
 428                        scale,
 429                        length,
 430                        taper_factor,
 431                        radius_outer,
 432                        radius_inner,
 433                        u,
 434                        v,
 435                        x,
 436                        y,
 437                        z,
 438                    )
 439                case _:
 440                    raise ValueError()
 441
 442        def peel_layer(s: str, parent: Branch | None) -> dict:
 443            head = s[0]
 444            head_branch = generate_branch(symbol=head, parent=parent)
 445            blist = patt.findall(s)  # should either find 0 or 2
 446            if len(blist) == 0:
 447                return {head_branch: {}}
 448            b1, b2 = blist
 449            d = {head_branch: [b1, b2]}
 450            for ind, b in enumerate(d[head_branch]):
 451                d[head_branch][ind] = peel_layer(b, head_branch)
 452            return d
 453
 454        d = peel_layer(s, None)
 455        # Flatten the dictionary into a list of objects
 456        # No two objects should be the same, so it should be OK to continue to use them as keys...
 457        lst: list[Branch] = []
 458        while len(d.keys()) > 0:
 459            keys = list(d.keys())
 460            for key in keys:
 461                lst.append(key)
 462                for val in d[key]:
 463                    d.update(val)  # unpack branches into root layer of dict
 464                del d[key]  # delete the old branch
 465        # First, print out the outer tubes and spheres
 466        ans = ""  # we will build the final string from this
 467
 468        # HOTFIX: preprocessing the file with pcpp takes a long time. It's single-threaded.
 469        # To speed this up, I'll not use these macros. I'll just directly generate the files with the values replaced.
 470        # ans += "#define OUTERRHO 1.0\n"
 471        # ans += "#define INNERRHO 0.1\n"
 472        # ans += "#define RELEVANCE relevance\n"
 473
 474        ans += "\n"
 475        ans += "// Outer Objects:\n"
 476        ans += "\n"
 477        gen: int = -1  # keep track of generation. When it increments, print a comment line saying so
 478        for b in lst:
 479            if b.generation > gen:
 480                gen = b.generation
 481                ans += "\n"
 482                ans += f"// Generation {gen:02d}\n"
 483            pos = (np.array(b.u) + np.array(b.v)) / 2
 484            if b.taper_factor == 1:
 485                ans += (
 486                    "{"
 487                    + f"[Cylinder: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r={b.radius_outer}mm axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=1.0 relevance"
 488                    + "}\n"
 489                )
 490                ans += (
 491                    "{"
 492                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_outer}mm] rho=1.0 relevance"
 493                    + "}\n"
 494                )
 495            else:
 496                ans += (
 497                    "{"
 498                    + f"[Cone: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r1={b.radius_outer / b.taper_factor}mm r2={b.radius_outer} axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=1.0 relevance"
 499                    + "}\n"
 500                )
 501                ans += (
 502                    "{"
 503                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_outer}mm] rho=1.0 relevance"
 504                    + "}\n"
 505                )
 506            ans += "\n"
 507        # Second, print out the inner tubes and spheres
 508        ans += "\n"
 509        ans += "// Inner Objects:\n"
 510        ans += "\n"
 511        gen: int = -1  # keep track of generation. When it increments, print a comment line saying so
 512        for b in lst:
 513            if b.generation > gen:
 514                gen = b.generation
 515                ans += "\n"
 516                ans += f"// Generation {gen:02d}\n"
 517            pos = (np.array(b.u) + np.array(b.v)) / 2
 518            if b.taper_factor == 1:
 519                ans += (
 520                    "{"
 521                    + f"[Cylinder: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r={b.radius_inner}mm axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=0.1 relevance"
 522                    + "}\n"
 523                )
 524                ans += (
 525                    "{"
 526                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_inner}mm] rho=0.1 relevance"
 527                    + "}\n"
 528                )
 529            else:
 530                ans += (
 531                    "{"
 532                    + f"[Cone: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r1={b.radius_inner / b.taper_factor}mm r2={b.radius_inner}mm axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=0.1 relevance"
 533                    + "}\n"
 534                )
 535                ans += (
 536                    "{"
 537                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_inner}mm] rho=0.1 relevance"

 263    @classmethod
 264    def _LungPhantom_phastring(  # ruff: ignore[complex-structure]
 265        cls,
 266        generation: int,
 267        scale_factor: float
 268        | None = None,  # Leave at None for default configuration
 269        taper: bool | None = None,  # Leave at None for default configuration
 270        fix: bool
 271        | None = None,  # Leave at None for default configuration. Turning this True will fix what appear to be errors in the original phantom. These errors are particularly noticeable with e.g. taper=True.
 272    ) -> str:
 273        """Generate the [Lung phantom](https://www.dkfz.de/en/x-ray-imaging-and-computed-tomography/lung-phantom) (by Markus Blank and Marc Kachelrieß) up to generation n."""
 274        if type(generation) is not int or generation < 2:
 275            msg = "'generation' must be an integer >=2"
 276            raise ValueError(msg)
 277        if scale_factor is None:
 278            scale_factor = 0.86  # default value is 0.86
 279        if taper is None:
 280            taper = False  # default phantom does not taper. It uses cylinders, not truncated cones.
 281        if fix is None:
 282            fix = False
 283        # Lindenmayer System
 284        s = "T[L[s][b]][R[s][b]]"
 285        for _ in range(generation - 2):
 286            s = s.replace("b", "B[X][Y]").replace("s", "S[X][Y]")
 287            s = s.replace("X", "b").replace("Y", "s")
 288        s = s.replace("b", "B").replace("s", "S")
 289        # Construct a dictionary tree of the terminal symbols T, L, R, B, S
 290        patt = regex.compile(r"(?V1)\[((?:[^\]\[]+|(?R))*+)\]")
 291
 292        @dataclasses.dataclass(frozen=True)
 293        class Branch:
 294            generation: int
 295            scale: float
 296            length: float
 297            taper_factor: float  # how much it shrinks by the end of the tube
 298            radius_outer: float
 299            radius_inner: float
 300            u: tuple[float, float, float]  # start point
 301            v: tuple[float, float, float]  # end point
 302            x: tuple[
 303                float, float, float
 304            ]  # defines the branch plane together with z
 305            y: tuple[
 306                float, float, float
 307            ]  # new branches will be rotated about y
 308            z: tuple[float, float, float]  # direction of branch
 309
 310        def generate_branch(symbol: str, parent: Branch | None) -> Branch:
 311            if parent is None:
 312                if symbol not in ("T", "L", "R"):
 313                    raise ValueError()
 314            match symbol:
 315                case "T":
 316                    return Branch(
 317                        0,
 318                        1.0,
 319                        100.0,
 320                        1,  # do not taper the trachea
 321                        9.0,
 322                        7.9,
 323                        (0.0, 0.0, 0.0),
 324                        (0.0, 0.0, 100.0),
 325                        (1.0, 0.0, 0.0),
 326                        (0.0, 1.0, 0.0),
 327                        (0.0, 0.0, 1.0),
 328                    )
 329                case "L":
 330                    return Branch(
 331                        1,
 332                        0.86,
 333                        34.4,
 334                        0.86 if taper else 1,
 335                        5.59,
 336                        4.644,
 337                        (0.0, 0.0, 100.0),
 338                        (28.1788, 0.0, 119.731),
 339                        (0.573576, 0.0, -0.819152),
 340                        (0.0, 1.0, 0.0),
 341                        (0.819152, 0.0, 0.573576),
 342                    )
 343                case "R":
 344                    return Branch(
 345                        1,
 346                        0.86,
 347                        43.0,
 348                        0.86 if taper else 1,
 349                        6.45,
 350                        5.504,
 351                        (0.0, 0.0, 100.0),
 352                        (-35.2235, 0.0, 124.664),
 353                        (-0.573576, 1.22461e-16, -0.819152),
 354                        (-7.02405e-17, -1.0, -1.00314e-16),
 355                        (-0.819152, 0.0, 0.573576),
 356                    )
 357                case "B":
 358                    assert parent is not None
 359                    r = scipy.spatial.transform.Rotation.from_rotvec(
 360                        np.array(parent.y) * -25, degrees=True
 361                    )
 362                    #
 363                    generation = parent.generation + 1
 364                    scale = parent.scale * scale_factor
 365                    length = scale * 50
 366                    # The amount to taper increase the radius toward the root
 367                    # depends on the radius of the parent.
 368                    # The parent can be the smaller of the two parents,
 369                    # in which case it needs to taper by less than the full
 370                    # inverse scale_factor
 371                    radius_outer = scale * 8.5 * (7.5 / 8.5 if fix else 1)
 372                    radius_inner = radius_outer * (7.4 / 8.5)
 373                    taper_factor = (
 374                        radius_outer / parent.radius_outer if taper else 1
 375                    )
 376                    u = parent.v
 377                    z = tuple(r.apply(parent.z))
 378                    x = tuple(
 379                        r.apply(parent.x)
 380                    )  # the web page isn't clear about how the new x should be determined
 381                    r2 = scipy.spatial.transform.Rotation.from_rotvec(
 382                        np.array(z) * 90,
 383                        degrees=True,  # I thought the angle here should have a minus sign, but no, it matches the 3rd generation example without it.
 384                    )
 385                    x = tuple(r2.apply(x))
 386                    v = tuple(np.array(u) + length * np.array(z))
 387                    y = tuple(np.cross(z, x))
 388                    return Branch(
 389                        generation,
 390                        scale,
 391                        length,
 392                        taper_factor,
 393                        radius_outer,
 394                        radius_inner,
 395                        u,
 396                        v,
 397                        x,
 398                        y,
 399                        z,
 400                    )
 401                case "S":
 402                    assert parent is not None
 403                    r = scipy.spatial.transform.Rotation.from_rotvec(
 404                        np.array(parent.y) * 45, degrees=True
 405                    )
 406                    #
 407                    generation = parent.generation + 1
 408                    scale = parent.scale * scale_factor
 409                    length = scale * 40
 410                    radius_outer = scale * 7.0 * (7.5 / 8.5 if fix else 1)
 411                    radius_inner = radius_outer * (5.9 / 7.0)
 412                    taper_factor = (
 413                        radius_outer / parent.radius_outer if taper else 1
 414                    )
 415                    u = parent.v
 416                    z = tuple(r.apply(parent.z))
 417                    x = tuple(
 418                        r.apply(parent.x)
 419                    )  # the web page isn't clear about how the new x should be determined
 420                    r2 = scipy.spatial.transform.Rotation.from_rotvec(
 421                        np.array(z) * 10,
 422                        degrees=True,  # I thought the angle here should have a minus sign, but no, it matches the 3rd generation example without it.
 423                    )
 424                    x = tuple(r2.apply(x))
 425                    v = tuple(np.array(u) + length * np.array(z))
 426                    y = tuple(np.cross(z, x))
 427                    return Branch(
 428                        generation,
 429                        scale,
 430                        length,
 431                        taper_factor,
 432                        radius_outer,
 433                        radius_inner,
 434                        u,
 435                        v,
 436                        x,
 437                        y,
 438                        z,
 439                    )
 440                case _:
 441                    raise ValueError()
 442
 443        def peel_layer(s: str, parent: Branch | None) -> dict:
 444            head = s[0]
 445            head_branch = generate_branch(symbol=head, parent=parent)
 446            blist = patt.findall(s)  # should either find 0 or 2
 447            if len(blist) == 0:
 448                return {head_branch: {}}
 449            b1, b2 = blist
 450            d = {head_branch: [b1, b2]}
 451            for ind, b in enumerate(d[head_branch]):
 452                d[head_branch][ind] = peel_layer(b, head_branch)
 453            return d
 454
 455        d = peel_layer(s, None)
 456        # Flatten the dictionary into a list of objects
 457        # No two objects should be the same, so it should be OK to continue to use them as keys...
 458        lst: list[Branch] = []
 459        while len(d.keys()) > 0:
 460            keys = list(d.keys())
 461            for key in keys:
 462                lst.append(key)
 463                for val in d[key]:
 464                    d.update(val)  # unpack branches into root layer of dict
 465                del d[key]  # delete the old branch
 466        # First, print out the outer tubes and spheres
 467        ans = ""  # we will build the final string from this
 468
 469        # HOTFIX: preprocessing the file with pcpp takes a long time. It's single-threaded.
 470        # To speed this up, I'll not use these macros. I'll just directly generate the files with the values replaced.
 471        # ans += "#define OUTERRHO 1.0\n"
 472        # ans += "#define INNERRHO 0.1\n"
 473        # ans += "#define RELEVANCE relevance\n"
 474
 475        ans += "\n"
 476        ans += "// Outer Objects:\n"
 477        ans += "\n"
 478        gen: int = -1  # keep track of generation. When it increments, print a comment line saying so
 479        for b in lst:
 480            if b.generation > gen:
 481                gen = b.generation
 482                ans += "\n"
 483                ans += f"// Generation {gen:02d}\n"
 484            pos = (np.array(b.u) + np.array(b.v)) / 2
 485            if b.taper_factor == 1:
 486                ans += (
 487                    "{"
 488                    + f"[Cylinder: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r={b.radius_outer}mm axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=1.0 relevance"
 489                    + "}\n"
 490                )
 491                ans += (
 492                    "{"
 493                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_outer}mm] rho=1.0 relevance"
 494                    + "}\n"
 495                )
 496            else:
 497                ans += (
 498                    "{"
 499                    + f"[Cone: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r1={b.radius_outer / b.taper_factor}mm r2={b.radius_outer} axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=1.0 relevance"
 500                    + "}\n"
 501                )
 502                ans += (
 503                    "{"
 504                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_outer}mm] rho=1.0 relevance"
 505                    + "}\n"
 506                )
 507            ans += "\n"
 508        # Second, print out the inner tubes and spheres
 509        ans += "\n"
 510        ans += "// Inner Objects:\n"
 511        ans += "\n"
 512        gen: int = -1  # keep track of generation. When it increments, print a comment line saying so
 513        for b in lst:
 514            if b.generation > gen:
 515                gen = b.generation
 516                ans += "\n"
 517                ans += f"// Generation {gen:02d}\n"
 518            pos = (np.array(b.u) + np.array(b.v)) / 2
 519            if b.taper_factor == 1:
 520                ans += (
 521                    "{"
 522                    + f"[Cylinder: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r={b.radius_inner}mm axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=0.1 relevance"
 523                    + "}\n"
 524                )
 525                ans += (
 526                    "{"
 527                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_inner}mm] rho=0.1 relevance"
 528                    + "}\n"
 529                )
 530            else:
 531                ans += (
 532                    "{"
 533                    + f"[Cone: x={pos[0]}mm y={pos[1]}mm z={pos[2]}mm l={b.length}mm r1={b.radius_inner / b.taper_factor}mm r2={b.radius_inner}mm axis({b.z[0]}mm, {b.z[1]}mm, {b.z[2]}mm)] rho=0.1 relevance"
 534                    + "}\n"
 535                )
 536                ans += (
 537                    "{"
 538                    + f"[Sphere: x={b.v[0]}mm y={b.v[1]}mm z={b.v[2]}mm r={b.radius_inner}mm] rho=0.1 relevance"
 539                    + "}\n"
 540                )
 541        return ans

 568    @classmethod
 569    def _DiscResolutionPhantom_phastring(  # ruff: ignore[complex-structure]
 570        cls,
 571        g1: int,  # positive integer
 572        g2: int,  # positive integer
 573        *,
 574        radius: float,
 575        rho: float,
 576        circularise: bool,
 577        alternate_with_g2: bool,
 578        randomise_parity: bool,  # effectively supersedes 'alternate_with_g2'
 579        seed: int | None = None,
 580    ) -> np.ndarray[tuple[int, ...], np.dtype[fv.Shape]]:  # ty: ignore[invalid-type-arguments]
 581        rng: np.random.Generator = np.random.default_rng(seed)
 582        shapes: list[np.ndarray[tuple[int, ...], np.dtype[fv.Shape]]] = []  # ty: ignore[invalid-type-arguments]
 583        s = np.array(0, dtype=fv.Shape)
 584        # ans: str = ""
 585        # Hardcoded parameters
 586        crf: float = np.sqrt(
 587            2 * (1 - np.cos(2 * np.pi / 6))
 588        )  # crf = cone radius factor
 589        sl: float = (
 590            radius * 1 / ((2 + 2 * (g1 - 1)) / 2) / (2 * g2 + 1)
 591        )  # sl = scale_lens
 592        if circularise:
 593            g2_genmax = 1 + math.ceil(2 * g2)
 594        else:
 595            g2_genmax = g2
 596        for g1_, g2_ in itertools.product(range(g1), range(g2_genmax)):
 597            r = sl * (1 + 2 * g1_) / 2 * np.cos(2 * np.pi / 6 / 2)
 598            l = sl * 2 / 2 * np.cos(2 * np.pi / 6 / 2)
 599            r1 = sl * (2 + 2 * g1_) / 2 * crf / 2
 600            r2 = sl * 2 * g1_ / 2 * crf / 2
 601            # Make corner translations
 602            tmag = radius * g2_ / (2 * g2 + 1) * np.cos(2 * np.pi / 6 / 2) * 2
 603            if g2_ == 0:
 604                tx = [0]
 605                ty = [0]
 606            else:
 607                tx = []
 608                ty = []
 609                angs = np.linspace(
 610                    0, 2 * np.pi, num=6, endpoint=False
 611                ).tolist()
 612                # Make the translations between corners
 613                for ang1, ang2 in itertools.pairwise([*angs, angs[0]]):
 614                    tx1 = np.cos(ang1)
 615                    ty1 = np.sin(ang1)
 616                    tx2 = np.cos(ang2)
 617                    ty2 = np.sin(ang2)
 618                    tx += (
 619                        tmag * np.linspace(tx1, tx2, endpoint=False, num=g2_)
 620                    ).tolist()
 621                    ty += (
 622                        tmag * np.linspace(ty1, ty2, endpoint=False, num=g2_)
 623                    ).tolist()
 624                if circularise:
 625                    for idx in reversed(range(len(tx))):
 626                        if (
 627                            np.sqrt(tx[idx] ** 2 + ty[idx] ** 2)
 628                            + radius / (2 * g2 + 1) * np.cos(2 * np.pi / 6 / 2)
 629                        ) > radius:
 630                            del tx[idx]
 631                            del ty[idx]
 632            for theta in (2 * np.pi / 6 * g1_) + np.linspace(
 633                0, 2 * np.pi, num=3, endpoint=False
 634            ):
 635                if alternate_with_g2:
 636                    theta += 0 if g2_ % 2 == 0 else np.pi  # ruff: ignore[redefined-loop-name]
 637                ax, ay, az = np.cos(theta), np.sin(theta), 0
 638                for tx_, ty_ in zip(tx, ty):
 639                    if randomise_parity:
 640                        theta += 0 if rng.random() < 0.5 else np.pi  # ruff: ignore[redefined-loop-name]
 641                    ax, ay, az = np.cos(theta), np.sin(theta), 0
 642                    # ans += (
 643                    #    "{"
 644                    #    + f"[Cone: x={-ax * r + tx_} y={-ay * r + ty_} z={-az * r} r1={r1} r2={r2} l={l} axis({ax},{ay},{az})] rho={rho}"
 645                    #    + "}\n"
 646                    # )
 647                    s_ = s.copy()
 648                    s_["rho"] = rho
 649                    s_["x"] = -ax * r + tx_
 650                    s_["y"] = -ay * r + ty_
 651                    s_["z"] = -az * r
 652                    s_["type"] = 2
 653                    s_["r1"] = r1
 654                    s_["r2"] = r2
 655                    s_["l"] = l
 656                    s_["axis"]["x"] = ax
 657                    s_["axis"]["y"] = ay
 658                    s_["axis"]["z"] = az
 659                    shapes.append(s_)
 660
 661        return np.array(shapes, dtype=fv.Shape)

 663    @classmethod
 664    def DiscResolutionPhantom(
 665        cls,
 666        generation: int,
 667        *,
 668        radius: float = 1,
 669        rho: float = 1,
 670        g2: int | None = None,
 671        circularise: bool = True,
 672        alternate_with_g2: bool = False,
 673        randomise_parity: bool = False,
 674        seed: int | None = None,
 675    ) -> FBVolume:
 676        """Generate a disc-shaped resolution phantom (use as a 2D phantom with z=0)."""
 677        if generation < 1:
 678            msg = "Generation should be a positive integer."
 679            raise ValueError(msg)
