 30def _phi(d: int, iterations: int = 100):  # ruff: ignore[missing-return-type-private-function]
 31    x = np.array(2, dtype=np.float64)
 32    for _ in range(iterations):
 33        x = np.float_power(1 + x, 1 / (d + 1))
 34    return x

 37def lds_nd(
 38    n: int,
 39    range_tuples: tuple[tuple[float, float], ...] | None = None,
 40) -> map[tuple[float, ...]]:
 41    """Generate a low-discrepancy sequence on an N-dimensional rectangular prism. Only really works for cubes."""
 42    if range_tuples is None:
 43        mins = tuple(0 for _ in range(n))
 44        maxs = tuple(1 for _ in range(n))
 45    else:
 46        if not len(range_tuples) == n:
 47            raise ValueError()
 48        mins, maxs = zip(*range_tuples)
 49    ranges = tuple(max_ - min_ for max_, min_ in zip(maxs, mins))
 50    biggest_range = max(ranges)
 51    ratios_wrt_biggest = tuple(range_ / biggest_range for range_ in ranges)
 52
 53    phi_n: np.float64 = _phi(n, 1000)
 54    as_: tuple[np.float64, ...] = tuple(
 55        np.float_power(phi_n, -i) for i in range(1, n + 1)
 56    )
 57
 58    filter_predicate = lambda zero_based_tuple: all(  # ruff: ignore[lambda-assignment]
 59        val < ratio_wrt_biggest
 60        for val, ratio_wrt_biggest in zip(zero_based_tuple, ratios_wrt_biggest)
 61    )
 62
 63    return map(  # ruff: ignore[unnecessary-map]
 64        lambda val_tuple: tuple(
 65            biggest_range * val + range_min
 66            for range_min, val in zip(mins, val_tuple)
 67        ),
 68        filter(
 69            filter_predicate,
 70            map(  # ruff: ignore[unnecessary-map]
 71                lambda m: tuple(((m * as_[i]) % 1) for i in range(n)),
 72                itertools.count(),  # generates 0,1,2,3,4,...
 73            ),
 74        ),
 75    )

 78def lds_cyl(
 79    zmin: float, zmax: float, rad: float, eccentricity: float = 1
 80) -> map[tuple[float, float]]:
 81    """Generate a low-discrepancy sequence on an unrolled cylinder."""
 82    return map(  # ruff: ignore[unnecessary-map]
 83        lambda zt: (zmin + zt[0] * eccentricity * rad, zt[1]),
 84        lds_nd(
 85            2,
 86            range_tuples=(
 87                (0, (zmax - zmin) / eccentricity / rad),
 88                (0, 2 * np.pi),
 89            ),
 90        ),
 91    )

 94def lds_cyl_tiled_batched_gen(
 95    R: float,  # cylinder radius  # ruff: ignore[invalid-argument-name]
 96    h: float,  # cylinder height
 97    zmin: float,  # cylinder min z
 98    N: int,  # points per batch  # ruff: ignore[invalid-argument-name]
 99    s: int = 1,  # parity of helix, -1 or 1
100) -> typing.Generator[list[tuple[float, float]]]:
101    """Create a generator of points.
102
103    Use like so:
104    g = make_point_generator(...)
105    # Then:
106    p1 = next(g) # p1 = (z1, phi1)
107    p2 = next(g) # p2 = (z2, phi2)
108    # ...
109    """  # ruff: ignore[docstring-missing-yields, docstring-missing-exception]
110    if s not in (-1, 1):
111        msg = f"s should be -1 or 1, but it is {s}"
112        raise ValueError(msg)
113
114    # twopi = 2*math.pi    # constant # Not used
115    phi2 = 1.324717957  # constant
116    n = 0
117
118    # m is the number of tiles around one sheared band
119    m = math.ceil(math.sqrt(N * 2 * math.pi * R / h))
120    # W is the physical sidelength of a square tile, before it is sheared
121    W = 2 * math.pi * R / m  # ruff: ignore[non-lowercase-variable-in-function]
122    # Stepsize
123    dz = s * W / m
124    dt = 2 * math.pi / m
125
126    while True:
127        # Generate next term in the 2D LDS sequence for the unit square
128        x1 = n / phi2 - math.floor(n / phi2)
129        x2 = n / phi2 / phi2 - math.floor(n / phi2 / phi2)
130        n += 1
131        # Add a shear to the square
132        x2 += s * x1 / m
133        # Calculate the bottom-left of the first shape
134        z: float = zmin + W * x2
135        t: float = 0.0 + W * x1 / R
136        if s > 0:
137            while z >= zmin:
138                z -= dz
139                t -= dt
140            ps = []
141            z += dz
142            t += dt
143            while z < zmin + h:
144                ps += [(z, t)]
145                z += dz
146                t += dt
147        else:  # opposite helix parity
148            while z < zmin + h:
149                z -= dz
150                t -= dt
151            ps = []
152            z += dz
153            t += dt
154            while z >= zmin:
155                ps += [(z, t)]
156                z += dz
157                t += dt
158        # Add an overall twist to the cylinder to make individual batches more evenly sample phi.
159        # It's unclear to me whether the added term should be positive or negative, depending on the parity 's'. A quick trial-and-error seems to yield good results when the sign is the negative of 's'. Otherwise, it appears that when the helical pitch is too high, then this interferes with the LDS and can tend to produce aligned points between batches or within batches.
160        ps = [
161            (z, t - s * (z - zmin) * (math.sqrt(5) - 1) / 2 / R) for z, t in ps
162        ]
163        yield ps

166def peel_cyl_points_arr(
167    zt_arr: np.ndarray | list[tuple[float, float]],
168    R: float,  # ruff: ignore[invalid-argument-name]
169    vZ: float,  # ruff: ignore[invalid-argument-name]
170    vA: float,  # ruff: ignore[invalid-argument-name]
171    phi_offset: float = 0,
172    reverse: bool = False,
173    reverse_parity: bool = False,
174    stop_time: float = 0,
175) -> np.ndarray:
176    """Reorder a list of evenly-sampled source points on the cylinder to give a close-to-fastest-possible traversal between them. Returns an array like np.array([[time1,z1,phi1], [time2,z2,phi2], ...])."""
177    # vZ = max speed of Z translation of the stage (units of distance per second).
178    # vA = max speed of the angular rotation of the stage (radians per second).
179    # reverse_parity changes the direction of the peel. (clockwise/anticlockwise)
180
181    # Initialise inputs
182    zt_arr = np.array(zt_arr)
183    if zt_arr.ndim != 2:
184        msg = "Input list/array has wrong dimensionality. It should be of shape (N,2)."
185        raise ValueError(msg)
186    if zt_arr.shape[1] != 2:
187        msg = "Input list/array is malformed. It should be of shape (N,2), but the length of the second axis was not 2."
188        raise ValueError(msg)
189
190    point_density: float = zt_arr.shape[0] / (
191        2 * np.pi * R * (zt_arr[:, 0].max() - zt_arr[:, 0].min())
192    )
193    peel_pitch = math.sqrt(3 / point_density * vZ / (R * vA))
194
195    zt_new = zt_arr.copy()
196    zs = zt_new[:, 0]
197    ts = zt_new[:, 1]
198    ts = np.mod(ts + phi_offset, 2 * np.pi, out=ts)
199    if reverse_parity:
200        zt_new[:, 1] = ts + 2 * np.pi * np.floor(
201            -zs / peel_pitch - ts / (2 * np.pi)
202        )
203    else:
204        zt_new[:, 1] = ts + 2 * np.pi * np.floor(
205            zs / peel_pitch - ts / (2 * np.pi)
206        )
207    # Sort by the ts
208    if reverse_parity:
209        zt_new = np.array(
210            sorted(
211                zt_new.tolist(),
212                key=(lambda zt_tup: zt_tup[1]),
213                reverse=True,
214            )
215        )
216    else:
217        zt_new = np.array(
218            sorted(zt_new.tolist(), key=(lambda zt_tup: zt_tup[1]))
219        )
220
221    if reverse:
222        zt_new = np.flip(zt_new, axis=0)
223
224    # Now compute the times between points. The first one is defined as t=0, marking the *beginning* of the first exposure.
225    dzdt = np.abs(np.diff(zt_new, axis=0))
226    dtimes = stop_time + np.amax(
227        dzdt / np.array([[vZ, vA]], dtype=np.float32), axis=1
228    )
229    times = np.array([0] + np.cumsum(dtimes).tolist())  # ruff: ignore[collection-literal-concatenation]
230
231    # Construct the full array
232    tzp = np.concatenate((times[:, np.newaxis], zt_new), axis=1)
233    return tzp  # ruff: ignore[unnecessary-assign]

236def peel_cyl_zts_gen(  # ruff: ignore[undocumented-public-function]
237    zt_gen: typing.Iterable[tuple[float, float]],
238    num_per_batch: int,
239    R: float,  # ruff: ignore[invalid-argument-name]
240    vZ: float,  # ruff: ignore[invalid-argument-name]
241    vA: float,  # ruff: ignore[invalid-argument-name]
242    phi_offset: float = 0,
243    reverse: bool = False,
244    reverse_parity: bool = False,
245    stop_time: float = 0,
246) -> typing.Generator[tuple[float, float, float]]:
247    zt_arrs = map(np.array, itertools.batched(zt_gen, num_per_batch))
248    rv = reverse
249    rp = reverse_parity
250
251    # Need a time offset between batches. Needs to include the time taken to get from the end of one batch to the beginning of the next.
252    t_offset: float | None = None
253    zlast, philast = None, None
254    zfirst, phifirst = None, None
255
256    for zt_arr in zt_arrs:
257        peeled_tzp = peel_cyl_points_arr(
258            zt_arr,
259            R=R,
260            vZ=vZ,
261            vA=vA,
262            phi_offset=phi_offset,
263            reverse=rv,
264            reverse_parity=rp,
265            stop_time=stop_time,
266        )
267        for time, z, phi in peeled_tzp:
268            if zfirst is None:  # Calculate the t_offset of this batch
269                zfirst, phifirst = z, phi
270                if t_offset is None:
271                    t_offset = 0
272                else:
273                    t_offset += stop_time + max(
274                        abs(zfirst - zlast) / vZ, abs(phifirst - philast) / vA
275                    )
276            time += t_offset  # ruff: ignore[redefined-loop-name]
277            yield (time, z, phi)
278            # phi_last = t  # Commented out because I'm not sure how best to start the next cycle when the previous one ends. Not too important, anyway.
279        zlast, philast = z, phi
280        zfirst, phifirst = None, None
281        # Reverse the parity and the order so that the next round starts where the previous ended, and the direction of rotation remains the same. It's the Z coordinate that bops up and down.
282        rv = not rv
283        rp = not rp
284    return

287def lds_cyl_dynamic_batched_gen(
288    zmin: float,
289    zmax: float,
290    R: float,  # The source-to-axis distance.  # ruff: ignore[invalid-argument-name]
291    vZ: float,  # Speed of translation of stage (units of distance per unit of time)  # ruff: ignore[invalid-argument-name]
292    vA: float,  # Speed of rotation of stage (radians per unit of time)  # ruff: ignore[invalid-argument-name]
293    num_per_batch: int,  # Number of points per complete sampling of the cylinder.
294    yield_whole_batches: bool = False,
295    theta_start: float = 0,
296    stop_time: float = 0,
297) -> (
298    typing.Generator[tuple[float, float, float]] | typing.Generator[np.ndarray]
299):
300    """Return a generator of source points which follows a close-to-fastest traversal between points. Assumes that the rotational and translation degrees of freedom are motorised separately. Imposes a minimum number of projections per 'round' around the cylinder, which is num_per_batch. Uses Murdock's 2D LDS over the cylinder."""
301    if num_per_batch < 2:
302        msg = "num_per_batch must be at least 2."
303        raise ValueError(msg)
304
305    # Define the point iterator
306    generator = lds_cyl(zmin=zmin, zmax=zmax, rad=R)
307    # Add the theta_start
308    generator = map(  # ruff: ignore[unnecessary-map]
309        lambda zt_tup: (zt_tup[0], zt_tup[1] + theta_start), generator
310    )
311    # Reorder the points within each batch according to the 'peel' method.
312    generator = peel_cyl_zts_gen(
313        generator,
314        num_per_batch=num_per_batch,
315        R=R,
316        vZ=vZ,
317        vA=vA,
318        phi_offset=0,
319        reverse=False,
320        reverse_parity=False,
321        stop_time=stop_time,
322    )
323    if yield_whole_batches:
324        generator = map(list, itertools.batched(generator, num_per_batch))
325
326    return generator  # ty:ignore[invalid-return-type]

329# @article{leopardi2006partition,
330#   title={A partition of the unit sphere into regions of equal area and small diameter},
331#   author={Leopardi, Paul},
332#   journal={Electronic Transactions on Numerical Analysis},
333#   volume={25},
334#   number={12},
335#   pages={309--327},

338def points_by_equal_area_partitioning_on_sphere(  # ruff: ignore[complex-structure, undocumented-public-function]
339    N: int,  # ruff: ignore[invalid-argument-name]
340    _take_half: bool = False,
341) -> tuple[np.ndarray, np.ndarray]:  # Theta, Phi
342    if type(N) is not int:
343        msg = "'n_odd' should be of type 'int'."
344        raise ValueError(msg)
345
346    # d = 2, because we work with the 2-dimensional sphere.
347
348    # eq. (2.9)
349    def Nu(theta: float) -> float:  # ruff: ignore[invalid-function-name]
350        return 4 * np.pi * np.square(np.sin(theta / 2))
351
352    # eq. (2.10) -- Theta is the inverse of Nu.
353    def Theta(x: float) -> float:  # ruff: ignore[invalid-function-name]
354        return 2 * np.arcsin(np.sqrt(x / (4 * np.pi)))
355
356    # -- Formula in Section 3 --
357
358    # 1. Determining the colatitudes of polar caps.
359    # Unnumbered equation before (3.1)
360    Nu_R: float = 4 * np.pi / N  # ruff: ignore[non-lowercase-variable-in-function]
361    # eq. (3.1)
362    theta_c: float = Theta(Nu_R)
363
364    # 2. Determining an ideal collar angle.
365    # Unnumbered equation
366    delta_I: float = np.sqrt(Nu_R)  # ruff: ignore[non-lowercase-variable-in-function]
367
368    # 3. Determining an ideal number of collars.
369    # Unnumbered equation
370    n_I: float = (  # ruff: ignore[non-lowercase-variable-in-function]
371        np.pi - 2 * theta_c
372    ) / delta_I  # ideal number of collars
373
374    # 4. Determining the actual number of collars.
375    # Unnumbered equation
376    n: int = 0 if (N == 2) else max(1, round(n_I))  # actual number of collars
377    if _take_half:
378        # It looks a bit better (more random, less structured) if the number of collars is odd
379        # n = n + (1 - n%2)
380        # I want to remove the perfect equatorial band of points, which may produce a weird artefact for the z frequencies
381        # To that end, I ensure that the number of collars is even here.
382        n = n + (n % 2)
383
384    # 5. Creating a list of the ideal number of regions in each collar.
385    delta_F: float = n_I / n * delta_I  # ruff: ignore[non-lowercase-variable-in-function]
386    theta_F: list[float] = list(theta_c + np.arange(n + 1) * delta_F)  # ruff: ignore[non-lowercase-variable-in-function]
387    y: list[float] = [
388        (Nu(theta_F[i + 1]) - Nu(theta_F[i])) / Nu_R for i in range(n)
389    ]
390
391    # 6. Creating a list of the actual number of regions in each collar.
392    a: list[float] = [0]
393    m: list[int] = []
394    for i in range(len(y)):
395        m.append(round(y[i] + a[i]))
396        a.append(a[-1] + y[i] - m[i])
397
398    if _take_half:
399        # Round up the m to make sure they're even
400        m = [mel + (mel % 2) for mel in m]
401        Nu_R = 4 * np.pi / (2 + sum(m))  # ruff: ignore[non-lowercase-variable-in-function]
402
403    # 7. Creating a list of colatitudes of each zone.
404    m_cumsum: list[float] = [0] + list(np.cumsum(m))  # ruff: ignore[collection-literal-concatenation]
405    theta: list[float] = [0]
406    theta.extend([Theta((1 + m_cumsum[i]) * Nu_R) for i in range(n + 1)])
407    if not _take_half:
408        theta.append(np.pi)
409
410    # --------------------------
411
412    # The way that the theta are defined is weird.
413    # Rewriting theta here to give the polar angles of the centers of each collar
414    theta: list[float] = [0]
415    theta.extend(
416        [
417            Theta(((1 + m_cumsum[i]) + (1 + m_cumsum[i + 1])) / 2 * Nu_R)
418            for i in range(n)
419        ]
420    )
421    if not _take_half:
422        theta.append(np.pi)
423
424    # Create a top and bottom "collar" corresponding to the polar caps.
425    m = [1] + m  # ruff: ignore[collection-literal-concatenation]
426    if not _take_half:
427        m.append(1)
428
429    # Define the offsets to the phi. These are the rotations of each collar, which are free to be chosen.
430    # phi_offset: np.ndarray = np.zeros((len(m),), dtype=np.float32)
431    if _take_half:
432        # the phi_offset needs to be an even function of (np.pi/2 - theta)
433        theta_el_abs = np.abs(np.pi / 2 - np.array(theta))
434        phi_offset = theta_el_abs * (1 + np.sqrt(5)) / 2
435    else:
436        phi_offset: np.ndarray = (
437            np.arange(len(m), dtype=np.float32) * (1 + np.sqrt(5)) / 2
438        )
439
440    # num_points: int = np.sum(m)
441    # Theta_out: np.ndarray = np.empty((num_points,), dtype=np.float32)
442    # Phi_out: np.ndarray = np.empty((num_points,), dtype=np.float32)
443
444    Theta_Phi: list[list[float]] = []  # ruff: ignore[non-lowercase-variable-in-function]
445    for theta_, phi_offset_, m_ in zip(theta, phi_offset, m):
446        phi_lst: np.ndarray = phi_offset_ + np.linspace(
447            0, 2 * np.pi, endpoint=False, num=m_
448        )
449        if _take_half:
450            phi_lst = phi_lst[: (1 if m_ == 1 else m_ // 2)]
451        Theta_Phi.extend([[theta_, phi_] for phi_ in phi_lst])
452
453    Theta_Phi_out = np.array(Theta_Phi)  # ruff: ignore[non-lowercase-variable-in-function]
454
455    return Theta_Phi_out[:, 0], Theta_Phi_out[:, 1]

474# From https://arxiv.org/pdf/0912.4540.pdf
475# The Roberts term is from https://extremelearning.com.au/how-to-evenly-distribute-points-on-a-sphere-more-effectively-than-the-canonical-fibonacci-lattice/#more-3069
476def uniformly_distributed_points_on_sphere_fibonacci_lattice(  # ruff: ignore[undocumented-public-function]
477    n_odd: int, roberts_term: bool = False
478) -> tuple[np.ndarray, np.ndarray]:  # Theta, Phi
479    if type(n_odd) is not int:
480        msg = "'n_odd' should be of type 'int'."
481        raise ValueError(msg)
482    # Must be odd. Eq. (2)
483    if n_odd % 2 == 0:
484        msg = "'n_odd' must be odd."
485        raise ValueError(msg)
486    # The Roberts term is appropriate for minimising the distance between points, whereas I think without the Roberts term it's better for having about the same area 'occupied' by each point. It's better without the Roberts term when we're trying to get the number of points to be roughly the same per unit solid angle (steradian), anywhere on the sphere.
487    P = n_odd  # ruff: ignore[non-lowercase-variable-in-function]
488    N = P // 2  # ruff: ignore[non-lowercase-variable-in-function]
489    golden_ratio = (1 + np.sqrt(5)) / 2
490    arange = np.arange(-N, N + 1)
491    Theta = np.arccos(2 * (arange + (0.36 if roberts_term else 0)) / P)  # ruff: ignore[non-lowercase-variable-in-function]
492    Phi = 2 * np.pi * arange / golden_ratio  # ruff: ignore[non-lowercase-variable-in-function]
493    # Normalise to the conventional coordinate ranges
494    Phi = np.mod(Phi, 2 * np.pi)  # ruff: ignore[non-lowercase-variable-in-function]
495    return Theta, Phi

179    # vA = max speed of the angular rotation of the stage (radians per second).
180    # reverse_parity changes the direction of the peel. (clockwise/anticlockwise)
181
182    # Initialise inputs
183    zt_arr = np.array(zt_arr)
184    if zt_arr.ndim != 2:

189        raise ValueError(msg)

192        2 * np.pi * R * (zt_arr[:, 0].max() - zt_arr[:, 0].min())
193    )
194    peel_pitch = math.sqrt(3 / point_density * vZ / (R * vA))
195
196    zt_new = zt_arr.copy()
197    zs = zt_new[:, 0]
198    ts = zt_new[:, 1]
199    ts = np.mod(ts + phi_offset, 2 * np.pi, out=ts)
200    if reverse_parity:
201        zt_new[:, 1] = ts + 2 * np.pi * np.floor(
202            -zs / peel_pitch - ts / (2 * np.pi)
203        )
204    else:
205        zt_new[:, 1] = ts + 2 * np.pi * np.floor(
206            zs / peel_pitch - ts / (2 * np.pi)
207        )
208    # Sort by the ts
209    if reverse_parity:

211            sorted(

214                reverse=True,

216        )

219            sorted(zt_new.tolist(), key=(lambda zt_tup: zt_tup[1]))
220        )
221
222    if reverse:
223        zt_new = np.flip(zt_new, axis=0)
224
225    # Now compute the times between points. The first one is defined as t=0, marking the *beginning* of the first exposure.
226    dzdt = np.abs(np.diff(zt_new, axis=0))
227    dtimes = stop_time + np.amax(
